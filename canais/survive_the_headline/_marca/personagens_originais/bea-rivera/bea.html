Incorporar em outro projeto

Modelo editável com todas as partes nomeadas: assets/bea-model.svg. Abre no Illustrator, Figma ou Inkscape.

<link rel="stylesheet" href="bea.css">
<div id="palco" style="width:480px;height:480px"></div>
<script src="bea.js"></script>
<script>
  const bea = new BeaRivera('#palco');
  bea.setExpression('curious').setPose('asking_question').setDirection('right');

  // Acessórios: true liga, false desliga, 'auto' deixa a pose decidir
  bea.setAccessory('backpack', true);
  bea.setAccessory('teddy_bear', true);
  bea.setAccessory('teddy_bear', 'auto');

  // Braço independente da pose
  bea.setArm('right', { shoulder: -60, elbow: -40, hand: 'open' });
  bea.setArm('right', null);
</script>
