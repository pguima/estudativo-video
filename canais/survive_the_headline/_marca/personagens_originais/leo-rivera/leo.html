Incorporar em outro projeto

Modelo editável com todas as partes nomeadas: assets/leo-model.svg. Abre no Illustrator, Figma ou Inkscape.

<link rel="stylesheet" href="leo.css">
<div id="palco" style="width:480px;height:480px"></div>
<script src="leo.js"></script>
<script>
  const leo = new LeoRivera('#palco');
  leo.setExpression('confident').setPose('using_phone').setDirection('right');

  // Acessórios: true liga, false desliga, 'auto' deixa a pose decidir
  leo.setAccessory('headphones', true);
  leo.setAccessory('smartphone', false);
  leo.setAccessory('smartphone', 'auto');

  // Braço independente da pose
  leo.setArm('right', { shoulder: -60, elbow: -40, hand: 'open' });
  leo.setArm('right', null);
</script>
