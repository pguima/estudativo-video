Incorporar em outro projeto

Modelo editável com todas as partes nomeadas: assets/biscuit-model.svg. Abre no Illustrator, Figma ou Inkscape.

<link rel="stylesheet" href="biscuit.css">
<div id="palco" style="width:480px;height:480px"></div>
<script src="biscuit.js"></script>
<script>
  const biscuit = new BiscuitRivera('#palco');
  biscuit.setExpression('curious').setPose('sitting').setDirection('right');

  // Acessórios: true liga, false desliga, 'auto' deixa a pose decidir
  biscuit.setAccessory('bone', true);
  biscuit.setAccessory('collar', false);
  biscuit.setAccessory('ball', 'auto');

  // Animações independentes da pose
  biscuit.setAnimation('tail_wag');
  biscuit.setAnimation('bark');
</script>
